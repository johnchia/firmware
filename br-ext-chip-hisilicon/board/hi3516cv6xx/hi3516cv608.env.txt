# U-Boot environment for the Hi3516CV608 on the OEM bootloader.
#
# This REPLACES the OEM environment, and the OEM boot path with it:
# bootcmd=sdupgrade;sysrecover;load_kernel reaches the kernel through an XRS
# descriptor written after the squashfs, which this tree does not produce. The
# bootcmd below reads the FIT from its own partition instead, which is what
# lets the ordinary kernel/rootfs split work.
#
# ethaddr is deliberately ABSENT. rcS's ethaddr_provision() fills it on first
# boot from a per-unit id -- flash uid, then a legacy uidraw, then the SoC die
# id through ipctool -- and persists it with fw_setenv, so the address is
# stable across reflashes. Writing a literal here would clone one MAC onto
# every board built from this file. A camera that still has the OEM ethaddr
# keeps it: ethaddr_provision only acts on an empty value.
#
# Same for the OEM's boardtype, boardver, hwcfg_* and ptzsupport keys. They
# describe the H4-52POX-S to OEM software that is not in this image, and
# carrying them here would state them for boards that are not that model.
baudrate=115200
bootdelay=1
loadaddr=0x42080000
baseaddr=0x42080000
soc=hi3516cv608
board=hi3516cv608
board_name=hi3516cv608
vendor=hisilicon
# mem= bounds the DEMO-board DTB's memory node, which claims 0xC0000000 (3 GB)
# on a 64 MB part. The OEM's own value was 41776K and it is NOT used here,
# because load_hisilicon cannot read it. Its parser is
# awk '/^mem=[0-9]+M/' -- M and only M -- so a size in K does not match, and
# the fallbacks behind it are fw_printenv osmem (unset) and then a literal 32.
# That 32 has no relationship to this board: it puts the MMZ at 0x42000000
# while the kernel given 41776K owns memory to 0x428CBFFF, so the video
# pipeline would be handed a zone starting 8.8 MiB inside the kernel's own RAM,
# with nothing logged. Spelling the size in M is what makes the script's answer
# a reading of the cmdline instead of a default that happens to be wrong.
#
# The split is 36/28: Linux takes 36 MiB and the MMZ gets 0x42400000..
# 0x44000000, ending exactly on the top of a part that U-Boot fixes the DTB
# memory node down to 0x04000000.
#
# It started as 32/32, borrowed from the EV200 because that was the only real
# datum for a 64 MB HiSilicon board. This board has now been measured and the
# borrowed number was wrong in the direction that matters. Peak MMZ demand
# under the raptor pipeline is 23,940 KiB, so 32 MiB of zone was 8 MiB of slack
# nothing was ever going to use -- while Linux, left 26.9 MiB of usable RAM
# after overhead, ran out: the OOM killer took rsd during a run (rvd survives
# it, carrying oom_score_adj -1000; rsd does not), and the board rebooted on
# its own, which is what panic=20 does with an OOM escalation.
#
# 28 MiB of zone keeps ~4 MiB over the measured peak and hands the 4 MiB back
# to the half that was actually short. The MMZ is still the easier knob to
# retune -- it is one line and a reboot -- but it is no longer the tight one.
#
# mmz_allocator=ot is on the cmdline for two reasons. It is correct -- this
# kernel reserves no CMA, so the cma branch would have nothing to allocate
# from -- and it stops check_allocator, which otherwise finds no allocator,
# sees totalmem < 128 and calls fw_setenv on EVERY boot, erasing the 64 KiB env
# sector each time for a value that never changes.
#
# Do NOT add mmz= here. insert_osal reads that key and switches to the cma
# branch, which is the opposite of what this board wants.
bootargs=mem=36M mmz_allocator=ot console=ttyAMA0,115200 panic=20 root=/dev/mtdblock3 rootfstype=squashfs init=/init mtdparts=sfc:192K(uboot),64K(env),2048K(kernel),5056K(rootfs),768K(rootfs_data),64K(devinfo)
# Stated rather than defaulted to: mem_total is the last value load_hisilicon
# would otherwise take from a literal (64), and it is right here only by luck.
totalmem=64M
# 0x40000 is the head of the kernel partition (192K uboot + 64K env) and
# 0x200000 is its whole 2048K, not the FIT's length: the slot is a constant
# of the table while the FIT grows with the kernel, and reading the slot
# cannot read past a partition that ends where rootfs begins.
bootcmd=sf probe 0; sf read ${baseaddr} 0x40000 0x200000; bootm ${baseaddr}
ipaddr=192.168.1.10
netmask=255.255.255.0
gatewayip=192.168.1.1
serverip=192.168.1.254
