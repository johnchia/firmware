# U-Boot environment for the Hi3516CV608 raptor targets, on OpenIPC's U-Boot.
#
# This is the compiled-in default of OpenIPC/u-boot-hi3516cv6xx
# (include/configs/hi-common.h, at the commit the defconfig pins) with the
# values this board needs changed, and each change says why below. The U-Boot
# reads a valid environment off flash INSTEAD of that default, not on top of
# it, so every key has to be here: a verb left out is a verb the board does not
# have. Keep it in step with hi-common.h when the pin moves.
#
# The layout is OpenIPC's, and the kernel and rootfs are found through the
# same variables: bootnor reads ${kernsize} from ${kernaddr}, and bootargs is
# built from ${osmem}, ${mtdids}:${mtdparts} and ${extras}. So
# `fw_setenv osmem 40M` works here exactly as on any OpenIPC camera. The
# backslashes in bootargs are OpenIPC's: they keep those variables unexpanded
# until bootnor's setargs builds the command line.
#
# ethaddr is deliberately ABSENT. rcS's ethaddr_provision() fills it on first
# boot from a per-unit id -- flash uid, then a legacy uidraw, then the SoC die
# id through ipctool -- and persists it with fw_setenv, so the address is
# stable across reflashes. Writing a literal here would clone one MAC onto
# every board built from this file. U-Boot invents a random one for its own
# use when the key is empty, but only in memory: it never saves it, so rcS
# still finds the key empty.
arch=arm
cpu=armv7
baudrate=115200
bootdelay=1
baseaddr=0x42080000
loadaddr=0x42080000
kernaddr=0x50000
kernsize=0x200000
rootaddr=0x250000
rootsize=0x500000
# OpenIPC's own bootcmd is `setenv bootcmd ${bootnor}; sf probe 0; saveenv; run
# bootcmd`: a default that writes its expanded self to flash on the first
# boot. An environment that is already on flash has no need of that, and it
# would spend an erase of the env sector on the camera's first boot.
bootcmd=run bootnor
bootnor=run fwrecovery; sf probe 0; setenv setargs setenv bootargs ${bootargs}; run setargs; sf read ${baseaddr} ${kernaddr} ${kernsize}; bootm ${baseaddr}
ubnor=${fetchcmd} ${baseaddr} u-boot-${soc}-nor.bin && run ubwrite
ubwrite=sf probe 0; sf erase 0x0 ${kernaddr}; sf write ${baseaddr} 0x0 ${kernaddr}
fwupd=${fetchcmd} ${baseaddr} firmware.${soc} && run fwwrite
fwrecovery=if env exists bootfail; then run fwupd; sleep 5; reset; fi
fwwrite=sf probe 0; sf erase ${kernaddr} ${filesize}; sf write ${baseaddr} ${kernaddr} ${filesize}
nfsroot=/srv/nfs/hi3516cv608
bootargsnfs=mem=${osmem} console=ttyAMA0,115200 panic=20 root=/dev/nfs rootfstype=nfs ip=${ipaddr}:::255.255.255.0::eth0 nfsroot=${serverip}:${nfsroot},v3,nolock rw ${extras}
bootnfs=setenv setargs setenv bootargs ${bootargsnfs}; run setargs; tftpboot ${baseaddr} uImage.${soc}; bootm ${baseaddr}
sdload=setenv fetchcmd fatload mmc 0
norload=sf probe 0; sf read
fetchcmd=tftpboot
# osmem bounds the DEMO-board DTB's memory node, which claims 0xC0000000
# (3 GB) on a 64 MB part, and load_hisilicon reads the MMZ split back from
# the mem= it becomes. Its parser is awk '/^mem=[0-9]+M/' -- M and only M --
# so the size must be spelled in M.
#
# The split is 36/28, not OpenIPC's 32/32: Linux takes 36 MiB and the MMZ
# gets 0x42400000..0x44000000, ending exactly on the top of a part that
# U-Boot fixes the DTB memory node down to 0x04000000. 32/32 was measured
# wrong in the direction that matters. Peak MMZ demand under the raptor
# pipeline is 23,940 KiB, so 32 MiB of zone was 8 MiB of slack nothing was
# ever going to use -- while Linux, left 26.9 MiB of usable RAM after
# overhead, ran out: the OOM killer took rsd during a run (rvd survives it,
# carrying oom_score_adj -1000; rsd does not), and the board rebooted on its
# own, which is what panic=20 does with an OOM escalation. 28 MiB of zone
# keeps ~4 MiB over the measured peak.
osmem=36M
# Stated rather than defaulted to: mem_total is the last value load_hisilicon
# would otherwise take from a literal (64), and it is right here only by luck.
totalmem=64M
# mmz_allocator=ot reaches the command line through ${extras}, the hook
# OpenIPC's bootargs leaves for it. It is correct -- this kernel reserves no
# CMA, so the cma branch would have nothing to allocate from -- and it stops
# check_allocator, which otherwise finds no allocator on the command line and
# calls fw_setenv on EVERY boot, erasing the env sector each time for a value
# that never changes.
#
# Do NOT add mmz= here. insert_osal reads that key and switches to the cma
# branch, which is the opposite of what this board wants.
extras=mmz_allocator=ot
mtdids=sfc
mtdparts=${mtdids}:256k(boot),64k(env),2048k(kernel),5120k(rootfs),7168k@0x50000(firmware),-(rootfs_data)
bootargs=mem=\${osmem} console=ttyAMA0,115200 panic=20 root=/dev/mtdblock3 rootfstype=squashfs init=/init mtdparts=\${mtdids}:\${mtdparts} \${extras}
board=hi3516cv608
vendor=hisilicon
soc=hi3516cv608
socmodel=
ipaddr=192.168.1.108
netmask=255.255.255.0
gatewayip=192.168.1.1
serverip=192.168.1.107
