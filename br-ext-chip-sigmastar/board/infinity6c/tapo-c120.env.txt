# TP-Link Tapo C120 (SSC377) -- the U-Boot environment written to the env
# partition at 0x40000.
#
# host-uboot-tools turns this into images/uboot-env.bin with mkenvimage, and
# make_full_image.sh places it at 256K. Blank lines and lines starting with #
# are dropped by mkenvimage, so the commentary costs nothing.
#
# WHY THIS FILE EXISTS AT ALL. Everything below except the last line is a
# transcription of the environment U-Boot compiles in and would have written
# itself on the first boot of a board with an erased env sector. It is here
# only so that wlandev can be, and wlandev has to be baked because the radio
# on this board needs a GPIO driven before it appears -- so it cannot be
# detected, so /etc/wireless/detect will never find it, so something has to
# name the arm before S40network runs. A camera that can be reached over the
# network can be told with `fw_setenv wlandev ...` instead and does not need
# any of this.
#
# THE RISK THIS CARRIES. A stored environment with a good CRC REPLACES the
# compiled-in default; it does not merge with it. Every variable the boot path
# needs must therefore be in this file, and a key dropped here is a key the
# board does not have. That is why the transcription is complete rather than
# minimal, and why the values below are worth checking against
# include/configs/sstar-common.h after a U-Boot bump.
#
# Escaping is literal: mkenvimage copies bytes. The \${...} pairs are meant to
# reach the stored value with the backslash still on them -- that is how the
# default env carries them, and hush resolves them at run time through the
# two-step in cmdnor. Do not "fix" them.
#
# Deliberately absent: ethaddr. U-Boot derives it from the NOR part's unique ID
# when it is empty or still the vendor default, so leaving it out is what gives
# each unit an address of its own. Also absent: memlx, memsz and totalmem,
# which infinity6c/chip.c sets from detected DRAM on every boot -- on the 64MB
# part that is LX_MEM=0x4000000 with a 32MB MMA heap.

# --- flash layout ----------------------------------------------------------
#
# 256k(boot),64k(env),2048k(kernel),8192k(rootfs),-(rootfs_data) on a 16MB
# part. rootmtd must equal the rootfs cap the image was size-checked against
# (BR2_OPENIPC_ROOTFS_PART_KB) or an image that passed its build-time check
# still overruns the partition it is written to.
baseaddr=0x21000000
kernaddr=0x50000
kernsize=0x200000
rootaddr=0x250000
rootsize=0x800000
rootmtd=8192k
soc=ssc377

# --- boot ------------------------------------------------------------------
#
# bootcmd is the compiled-in default verbatim, self-rewriting and all: it sets
# bootcmd to the expanded cmdnor, saves, and runs it. Kept in that form rather
# than pre-expanded so a board flashed with this file takes exactly the path a
# board with an erased env would have taken.
bootargs=console=ttyS0,115200 panic=20 root=/dev/mtdblock3 init=/init mtdparts=NOR_FLASH:256k(boot),64k(env),2048k(kernel),\${rootmtd}(rootfs),-(rootfs_data) LX_MEM=\${memlx} mma_heap=mma_heap_name0,miu=0,sz=\${memsz} cma=2M
bootcmd=setenv bootcmd ${cmdnor}; sf probe 0; saveenv; run bootcmd
cmdnor=sf probe 0; setenv setargs setenv bootargs ${bootargs}; run setargs; sf read ${baseaddr} ${kernaddr} ${kernsize}; bootm ${baseaddr}

# --- the update verbs ------------------------------------------------------
#
# tftpboot by default; `run sdcard` switches them to a FAT card.
updatetool=tftpboot
sdcard=setenv updatetool fatload mmc 0
ubnor=${updatetool} ${baseaddr} u-boot-${soc}-nor.bin && run ubwrite
uknor=${updatetool} ${baseaddr} uImage.${soc} && run ukwrite
urnor=${updatetool} ${baseaddr} rootfs.squashfs.${soc} && run urwrite
ubwrite=sf probe 0; sf erase 0x0 ${kernaddr}; sf write ${baseaddr} 0x0 ${kernaddr}
ukwrite=sf probe 0; sf erase ${kernaddr} ${kernsize}; sf write ${baseaddr} ${kernaddr} ${filesize}
urwrite=sf probe 0; sf erase ${rootaddr} ${rootsize}; sf write ${baseaddr} ${rootaddr} ${filesize}

# --- radio -----------------------------------------------------------------
#
# Names the arm in /etc/wireless/usb that powers the RTL8188FTV through GPIO 42
# (PAD_FUART_RX), pulses its enable on GPIO 8 (PAD_UART1_TX) the way the vendor
# firmware does, then modprobes 8188fu. The defconfig header says where each
# pin comes from.
#
# Credentials are per unit and are not in this file. They arrive from the setup
# portal, which S41setup raises once wlan0 exists.
wlandev=rtl8188fu-ssc377-tapo-c120
