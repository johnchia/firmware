# Wyze Cam v3 (T31X) -- the U-Boot environment written to mtd1.
#
# host-uboot-tools turns this into images/uboot-env.bin with mkenvimage, and
# make_full_image_ingenic.sh places it at 0x50000. Blank lines and lines
# starting with # are dropped by mkenvimage, so the commentary costs nothing.
#
# The recovery verbs below -- autoupdate, loaduenv, overlay_wipe and the
# button_cmd_0 that runs it -- are transcribed from the environment read out
# of a working board. They are the only way back into this camera: it has no
# Ethernet PHY and no serial header, so if the bootloader comes up without a
# usable SD-card path the remaining options are USB DFU and a SPI clip.
# Change them only against a board you can reach another way.
#
# Escaping is literal: mkenvimage copies bytes, and the \" pairs are unescaped
# by hush at run time. Do not "fix" them.

# --- the SD-card recovery path --------------------------------------------
#
# autoupdate erases and rewrites the whole chip from autoupdate-full.bin on a
# FAT32 card, then stamps autoupdate-full.done beside it so the next boot
# skips it. loaduenv imports uenv.txt from the same card into the saved
# environment and stamps uenv.done -- which is how wifi credentials reach a
# camera that has no other interface. Delete a stamp to apply it again;
# set enable_updates=false to disable both.
enable_updates=true
# main_loop runs preboot before bootcmd and independently of it, so the
# SD-card reflash still happens even if bootcmd is broken. bootcmd runs
# autoupdate too; the second call sees the stamp and returns immediately.
preboot=run autoupdate
autoupdate=if test \"${enable_updates}\" = \"true\"; then echo \"checking for update file\"; if fatsize mmc 0:1 autoupdate-full.done; then echo \"AU: already applied\"; else if fatload mmc 0:1 ${loadaddr} autoupdate-full.bin; then echo \"AU: flashing autoupdate-full.bin\"; if sf probe && sf erase 0 ${flash_len} && sf write ${loadaddr} 0 ${filesize}; then fatwrite mmc 0:1 ${loadaddr} autoupdate-full.done 1; echo \"AU: done, rebooting\"; reset; fi; fi; fi; fi
loaduenv=if test \"${enable_updates}\" = \"true\"; then if fatsize mmc 0:1 uenv.done; then echo \"UENV: uenv.txt already applied\"; else if fatload mmc 0:1 ${loadaddr} uenv.txt; then echo \"UENV: importing uenv.txt\"; if env import -t -r ${loadaddr} ${filesize}; then saveenv; fatwrite mmc 0:1 ${loadaddr} uenv.done 1; echo \"UENV: env saved\"; fi; fi; fi; fi

# The reset button, read once early in main_loop by CONFIG_BUTTON_CMD against
# the gpio-keys node named "reset". overlay_wipe erases data_addr/data_size,
# which are read at run time -- so a layout change that misses them points the
# recovery button at the rootfs, and nobody finds out until they need it.
button_cmd_0=echo \"RST: factory reset\"; run overlay_wipe
button_cmd_0_name=reset
overlay_wipe=echo \"wiping overlay\"; sf probe && sf erase ${data_addr} ${data_size} && echo \"overlay wipe done\"

# --- flash layout ----------------------------------------------------------
#
# The first 448k belongs to the bootloader and cannot move: CONFIG_ENV_OFFSET
# 0x50000 and CONFIG_ENV_SIZE 0x10000 are compiled into U-Boot, fixing
# boot/env/backup. Everything past 0x70000 is ours, and these are OpenIPC's
# sizes -- 2048k kernel, 8192k rootfs, remainder to the overlay.
#
# The partition NAME matters as much as the offset: OpenIPC's /init finds the
# overlay by looking for rootfs_data in /proc/mtd. Called anything else --
# thingino names it data -- the mount fails, /init falls back to a tmpfs, and
# the camera silently forgets every change at the next reboot.
mtdparts=jz_sfc:320k(boot),64k(env),64k(backup),2048k(kernel),8192k(rootfs),5696k(rootfs_data),16384k@0(all)
flash_len=0x1000000
kern_addr=0x70000
kern_size=0x200000
data_addr=0xa70000
data_size=0x590000
root=/dev/mtdblock4
rootfstype=squashfs
init=/init

# --- memory ----------------------------------------------------------------
#
# T31X with 128 MB: 99 MB to Linux, 29 MB of ISP reserve at the top. Taken
# from a working board -- the ISP allocates out of rmem, so shrinking it
# shows up as capture failures rather than as an out-of-memory.
osmem=99M@0x0
rmem=29M@0x6300000

# --- console and boot ------------------------------------------------------
loadaddr=0x80600000
serialport=ttyS1
baudrate=115200
panic_timeout=2
# The three fatwrite calls are breadcrumbs, not function. With no serial
# header there is otherwise no way to tell a kernel that never started
# from one that started and died: ub1 says the recovery block completed,
# ub2 carries the first four bytes of what came off the SPI (it should
# read 27 05 19 56, the uImage magic, which proves the read and the
# stored image), and ub3 exists only if bootm returned -- meaning it
# refused the image rather than handing over. Separators are ";" and not
# "&&" so a missing card cannot stop the boot.
bootcmd=run autoupdate;run loaduenv;sf probe;setenv bootargs mem=${osmem} rmem=${rmem} nogmac console=${serialport},${baudrate}n8 panic=${panic_timeout} root=${root} rootfstype=${rootfstype} init=${init} mtdparts=${mtdparts};sf read ${loadaddr} ${kern_addr} ${kern_size};bootm ${loadaddr}

# --- radio -----------------------------------------------------------------
#
# The only interface this board has. wlandev names the bring-up sequence in
# /etc/wireless/sdio -- the GPIO 57 power cycle, the SDIO rescan and the
# modprobe -- and its presence is also what tells S40network and rcd that the
# camera is wireless. Credentials are per unit and are not in this file; they
# arrive from the SD card via loaduenv (see wyze-v3.uenv.txt) or from the
# setup portal.
wlandev=atbm603x-t31-wyze-v3
disable_sd=false
disable_eth=true
